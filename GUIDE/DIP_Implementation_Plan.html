💡 Plain-English Guide (What I Expect From You Here):

In this step, all I need from you is to write 3 straightforward Python functions in a file named src/dct_utils.py:

  1. Function 1: Open an image and chop it into a grid of tiny $8 \times 8$ pixel squares.
  2. Function 2: Convert those squares to frequencies (using standard scipy.fft.dctn) and zero out the high-frequency numbers. This mimics what JPEG compression does!
  3. Function 3: Take those frequency numbers and turn them back into a picture (using scipy.fft.idctn).

You don't have to write this from scratch—the full starter code is ready below for you to use and test!

What to Do:

Write three core functions that handle all classical image-to-frequency conversion:

Function 1: load_and_slice(image_path)

  • Purpose: Reads an image, converts to grayscale, crops dimensions to multiples of 8, and slices into $8 \times 8$ pixel blocks.
  • Library: cv2.imread(path, cv2.IMREAD_GRAYSCALE) from OpenCV.
  • Input: File path string (e.g., "datasets/Set5/baby.png").
  • Output: A NumPy array of shape (num_blocks, 8, 8) containing all pixel blocks.
# Pseudocode:
def load_and_slice(image_path):
    img = cv2.imread(image_path, cv2.IMREAD_GRAYSCALE)
    img = img.astype(np.float32) / 255.0  # Normalize to 0-1
    
    h, w = img.shape
    h, w = (h // 8) * 8, (w // 8) * 8  # Crop to multiple of 8
    img = img[:h, :w]
    
    blocks = []
    for r in range(0, h, 8):
        for c in range(0, w, 8):
            blocks.append(img[r:r+8, c:c+8])
    
    return np.array(blocks), (h, w)  # Return blocks + original dimensions

Function 2: apply_dct_and_truncate(blocks, cutoff=15)

  • Purpose: Applies 2D-DCT to each block, saves the full coefficient matrix (ground truth $Y$), and creates a truncated version (input $X$) by zeroing out high frequencies.
  • Library: scipy.fft.dctn(block, norm='ortho').
  • Cutoff Strategy: Use a zigzag scan order. Keep the first cutoff coefficients in zigzag order, set the rest to 0.
  • Output: Two arrays: X_truncated (input) and Y_full (ground truth), both shape (num_blocks, 64).
# Pseudocode:
from scipy.fft import dctn

def get_zigzag_indices():
    """Returns the zigzag scan order for an 8x8 block as a list of (row, col) pairs."""
    # Standard JPEG zigzag order for 8x8
    indices = []
    for s in range(15):  # sum of (row + col) goes from 0 to 14
        if s % 2 == 0:
            for i in range(min(s, 7), max(0, s - 7) - 1, -1):
                indices.append((i, s - i))
        else:
            for i in range(max(0, s - 7), min(s, 7) + 1):
                indices.append((i, s - i))
    return indices

def apply_dct_and_truncate(blocks, cutoff=15):
    zigzag = get_zigzag_indices()
    X_list, Y_list = [], []
    
    for block in blocks:
        dct_block = dctn(block, norm='ortho')
        Y_flat = dct_block.flatten()
        
        # Create truncated version
        X_block = np.zeros_like(dct_block)
        for idx in range(cutoff):
            r, c = zigzag[idx]
            X_block[r, c] = dct_block[r, c]
        X_flat = X_block.flatten()
        
        X_list.append(X_flat)
        Y_list.append(Y_flat)
    
    return np.array(X_list), np.array(Y_list)

Function 3: reconstruct_from_dct(dct_blocks_flat, image_shape)

  • Purpose: Takes a set of 64-length DCT coefficient vectors, reshapes each to $8 \times 8$, applies Inverse DCT (IDCT), and stitches the blocks back into a full image.
  • Library: scipy.fft.idctn(block, norm='ortho').
  • Output: A full 2D grayscale image array.
# Pseudocode:
from scipy.fft import idctn

def reconstruct_from_dct(dct_blocks_flat, image_shape):
    h, w = image_shape
    img = np.zeros((h, w), dtype=np.float32)
    idx = 0
    for r in range(0, h, 8):
        for c in range(0, w, 8):
            block_flat = dct_blocks_flat[idx]
            block_2d = block_flat.reshape(8, 8)
            img[r:r+8, c:c+8] = idctn(block_2d, norm='ortho')
            idx += 1
    return np.clip(img, 0, 1)  # Clip to valid pixel range

# src/dataset.py
import torch
from torch_geometric.data import Data, Dataset
from src.graph_builder import build_frequency_graph  # Your own file (Stage 2)

class DCTGraphDataset(Dataset):
    def __init__(self, X_blocks, Y_blocks, cutoff=15):
        """
        X_blocks: (N, 64) truncated DCT coefficients
        Y_blocks: (N, 64) full ground truth DCT coefficients
        cutoff: number of preserved low-frequency coefficients
        """
        super().__init__()
        self.X = torch.tensor(X_blocks, dtype=torch.float32)
        self.Y = torch.tensor(Y_blocks, dtype=torch.float32)
        self.cutoff = cutoff
        self.edge_index = build_frequency_graph()  # Built once, shared by all blocks

    def len(self):
        return len(self.X)

    def get(self, idx):
        x = self.X[idx]  # (64,) truncated
        y = self.Y[idx]  # (64,) full ground truth

        # Node features: [value, u_coord/8, v_coord/8]
        coords = torch.tensor([[i // 8, i % 8] for i in range(64)], dtype=torch.float32) / 8.0
        node_features = torch.cat([x.unsqueeze(1), coords], dim=1)  # (64, 3)

        # Missing mask: True for high-freq nodes that need prediction
        missing_mask = torch.zeros(64, dtype=torch.bool)
        missing_mask[self.cutoff:] = True

        return Data(
            x=node_features,
            edge_index=self.edge_index,
            y=y,
            missing_mask=missing_mask
        )

What to Do:

Write two standard image quality metric functions and run the baseline evaluation.

Function 1: calculate_psnr(original, reconstructed)

  • Library: skimage.metrics.peak_signal_noise_ratio from scikit-image, or manual calculation.
$$\text{PSNR} = 10 \cdot \log_{10} \left( \frac{\text{MAX}^2}{\text{MSE}} \right) \text{ dB}$$
Where MAX = 1.0 (for normalized images) and MSE is the mean squared pixel difference.

Function 2: calculate_ssim(original, reconstructed)

  • Library: skimage.metrics.structural_similarity.
  • Output: A score between 0 and 1 (1 = perfect match).
# src/metrics.py
import numpy as np
from skimage.metrics import peak_signal_noise_ratio, structural_similarity

def calculate_psnr(original, reconstructed):
    """Calculate PSNR between two images (both normalized 0-1)."""
    return peak_signal_noise_ratio(original, reconstructed, data_range=1.0)

def calculate_ssim(original, reconstructed):
    """Calculate SSIM between two images (both normalized 0-1)."""
    return structural_similarity(original, reconstructed, data_range=1.0)

File: src/graph_builder.py

What to Do:

Define the graph topology: how the 64 frequency nodes in an $8 \times 8$ DCT block are connected.

Edge Types to Build:

  1. Spatial Adjacency Edges: Connect each node $(u, v)$ to its direct frequency neighbors $(u+1, v)$ and $(u, v+1)$.
  2. Harmonic Edges: Connect lower frequencies to their harmonics: $(u, v) \rightarrow (2u, 2v)$ (if within the $8 \times 8$ grid).
  3. Bidirectional: All edges should be bidirectional (add both directions) so messages flow in both ways.
# src/graph_builder.py
import torch

def build_frequency_graph():
    """
    Build the edge_index tensor for a single 8x8 DCT frequency block.
    Returns: edge_index of shape (2, num_edges)
    """
    edges = []
    
    for u in range(8):
        for v in range(8):
            node_id = u * 8 + v  # Flatten 2D (u,v) to 1D index
            
            # 1. Spatial neighbor: right (u, v+1)
            if v + 1 < 8:
                neighbor_id = u * 8 + (v + 1)
                edges.append([node_id, neighbor_id])
                edges.append([neighbor_id, node_id])  # Bidirectional
            
            # 2. Spatial neighbor: down (u+1, v)
            if u + 1 < 8:
                neighbor_id = (u + 1) * 8 + v
                edges.append([node_id, neighbor_id])
                edges.append([neighbor_id, node_id])
            
            # 3. Diagonal neighbor: (u+1, v+1)
            if u + 1 < 8 and v + 1 < 8:
                neighbor_id = (u + 1) * 8 + (v + 1)
                edges.append([node_id, neighbor_id])
                edges.append([neighbor_id, node_id])
            
            # 4. Harmonic edge: (u, v) -> (2u, 2v) if in bounds
            hu, hv = 2 * u, 2 * v
            if hu < 8 and hv < 8:
                harmonic_id = hu * 8 + hv
                if harmonic_id != node_id:
                    edges.append([node_id, harmonic_id])
                    edges.append([harmonic_id, node_id])
    
    # Remove duplicate edges
    edge_set = list(set(tuple(e) for e in edges))
    edge_index = torch.tensor(edge_set, dtype=torch.long).t().contiguous()
    
    return edge_index  # Shape: (2, num_edges)

What to Do:

Define the Graph Neural Network model. Start simple with a 3-layer GCN (Graph Convolutional Network) with ReLU activations and a linear output head.

# src/models.py
import torch
import torch.nn as nn
from torch_geometric.nn import GCNConv

class FrequencyGNN(nn.Module):
    def __init__(self, in_channels=3, hidden_channels=128, num_layers=3):
        """
        in_channels: Node feature size (value + u_coord + v_coord = 3)
        hidden_channels: Width of hidden GNN layers
        num_layers: Number of GCN message-passing layers
        """
        super().__init__()
        
        self.convs = nn.ModuleList()
        self.convs.append(GCNConv(in_channels, hidden_channels))
        for _ in range(num_layers - 1):
            self.convs.append(GCNConv(hidden_channels, hidden_channels))
        
        # Final linear head: predict 1 scalar (DCT amplitude) per node
        self.predictor = nn.Linear(hidden_channels, 1)
        self.relu = nn.ReLU()

    def forward(self, x, edge_index):
        """
        x: Node features (num_nodes, 3)
        edge_index: Graph edges (2, num_edges)
        Returns: Predicted DCT values for all 64 nodes (num_nodes, 1)
        """
        for conv in self.convs:
            x = self.relu(conv(x, edge_index))
        
        out = self.predictor(x)  # (num_nodes, 1)
        return out.squeeze(-1)   # (num_nodes,)

# train.py - The Main Orchestrator
import os
import glob
import argparse
import numpy as np
import torch
import torch.nn as nn
from torch_geometric.loader import DataLoader

from src.dct_utils import load_and_slice, apply_dct_and_truncate
from src.dataset import DCTGraphDataset
from src.models import FrequencyGNN

def prepare_training_data(image_folder, cutoff=15):
    """Load all images, slice into blocks, apply DCT and truncate."""
    all_X, all_Y = [], []
    image_files = glob.glob(os.path.join(image_folder, "*.jpg"))
    
    print(f"Processing {len(image_files)} images from {image_folder}...")
    for img_path in image_files:
        blocks, shape = load_and_slice(img_path)
        X, Y = apply_dct_and_truncate(blocks, cutoff=cutoff)
        all_X.append(X)
        all_Y.append(Y)
    
    all_X = np.concatenate(all_X, axis=0)
    all_Y = np.concatenate(all_Y, axis=0)
    print(f"Total training blocks: {len(all_X)}")
    return all_X, all_Y

def train(args):
    device = torch.device(args.device if torch.cuda.is_available() else 'cpu')
    print(f"Training on: {device}")
    
    # 1. Prepare data
    X_train, Y_train = prepare_training_data(args.train_folder, cutoff=args.cutoff)
    dataset = DCTGraphDataset(X_train, Y_train, cutoff=args.cutoff)
    dataloader = DataLoader(dataset, batch_size=args.batch_size, shuffle=True)
    
    # 2. Create model
    model = FrequencyGNN(in_channels=3, hidden_channels=args.hidden, num_layers=args.layers).to(device)
    optimizer = torch.optim.Adam(model.parameters(), lr=args.lr)
    criterion = nn.MSELoss()
    
    # 3. Training loop
    for epoch in range(args.epochs):
        total_loss = 0
        num_batches = 0
        
        for batch in dataloader:
            batch = batch.to(device)
            optimizer.zero_grad()
            
            # Forward pass: predict all 64 nodes
            predictions = model(batch.x, batch.edge_index)
            
            # Loss: only on missing high-frequency nodes
            loss = criterion(
                predictions[batch.missing_mask],
                batch.y[batch.missing_mask]
            )
            
            # Backward pass
            loss.backward()
            optimizer.step()
            
            total_loss += loss.item()
            num_batches += 1
        
        avg_loss = total_loss / num_batches
        print(f"Epoch [{epoch+1}/{args.epochs}]  Loss: {avg_loss:.6f}")
    
    # 4. Save trained model
    os.makedirs("checkpoints", exist_ok=True)
    save_path = "checkpoints/model.pth"
    torch.save(model.state_dict(), save_path)
    print(f"Model saved to {save_path}")

if __name__ == "__main__":
    parser = argparse.ArgumentParser()
    parser.add_argument("--train_folder", type=str, default="datasets/BSDS500/train")
    parser.add_argument("--epochs", type=int, default=30)
    parser.add_argument("--batch_size", type=int, default=64)
    parser.add_argument("--lr", type=float, default=1e-3)
    parser.add_argument("--hidden", type=int, default=128)
    parser.add_argument("--layers", type=int, default=3)
    parser.add_argument("--cutoff", type=int, default=15)
    parser.add_argument("--device", type=str, default="cuda")
    args = parser.parse_args()
    train(args)

# evaluate.py
import os
import glob
import numpy as np
import torch
import matplotlib.pyplot as plt

from src.dct_utils import load_and_slice, apply_dct_and_truncate, reconstruct_from_dct
from src.metrics import calculate_psnr, calculate_ssim
from src.graph_builder import build_frequency_graph
from src.models import FrequencyGNN

def evaluate_image(model, img_path, cutoff=15, device='cpu'):
    """Evaluate a single image: baseline vs GNN reconstruction."""
    import cv2
    
    # Load and process
    blocks, shape = load_and_slice(img_path)
    X_trunc, Y_full = apply_dct_and_truncate(blocks, cutoff=cutoff)
    
    # Baseline reconstruction (zero-padded IDCT)
    baseline_img = reconstruct_from_dct(X_trunc, shape)
    
    # GNN prediction
    edge_index = build_frequency_graph().to(device)
    model.eval()
    predicted_blocks = []
    
    with torch.no_grad():
        for i in range(len(X_trunc)):
            x_flat = torch.tensor(X_trunc[i], dtype=torch.float32)
            coords = torch.tensor([[j // 8, j % 8] for j in range(64)], dtype=torch.float32) / 8.0
            node_feat = torch.cat([x_flat.unsqueeze(1), coords], dim=1).to(device)
            
            pred = model(node_feat, edge_index).cpu().numpy()
            
            # Combine: keep known low freqs, use GNN for high freqs
            combined = X_trunc[i].copy()
            combined[cutoff:] = pred[cutoff:]
            predicted_blocks.append(combined)
    
    predicted_blocks = np.array(predicted_blocks)
    gnn_img = reconstruct_from_dct(predicted_blocks, shape)
    
    # Load original
    original = cv2.imread(img_path, cv2.IMREAD_GRAYSCALE).astype(np.float32) / 255.0
    h, w = shape
    original = original[:h, :w]
    
    # Metrics
    baseline_psnr = calculate_psnr(original, baseline_img)
    baseline_ssim = calculate_ssim(original, baseline_img)
    gnn_psnr = calculate_psnr(original, gnn_img)
    gnn_ssim = calculate_ssim(original, gnn_img)
    
    return {
        'original': original,
        'baseline': baseline_img,
        'gnn': gnn_img,
        'baseline_psnr': baseline_psnr,
        'baseline_ssim': baseline_ssim,
        'gnn_psnr': gnn_psnr,
        'gnn_ssim': gnn_ssim,
        'psnr_gain': gnn_psnr - baseline_psnr,
    }

if __name__ == "__main__":

def plot_comparison(result, image_name, show_popup=True):

def plot_error_heatmap(result, image_name, show_popup=True):
    """Plot pixel-wise error heatmap: Baseline vs GNN."""
    fig, axes = plt.subplots(1, 2, figsize=(12, 5))
    
    error_baseline = np.abs(result['original'] - result['baseline'])
    error_gnn = np.abs(result['original'] - result['gnn'])
    
    im1 = axes[0].imshow(error_baseline, cmap='hot', vmin=0, vmax=0.15)
