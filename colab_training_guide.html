Google Colab Training Guide

Complete cell-by-cell script to train the Dual-Domain Hybrid Model on DIV2K + Flickr2K (DF2K) with 25 GNN Epochs and 60 CNN Epochs.

⚡ GPU: T4 or A100 📦 Dataset: DF2K (3,450 2K/4K Images) 🧠 GNN: 25 Epochs 🖼️ Spatial CNN: 60 Epochs 💾 Output: checkpoints/hybrid_model.pth

🚀 Quick Setup Instructions for Colab

  1. Open Google Colab and click New Notebook.
  2. Set the hardware accelerator: Navigate to Runtime → Change runtime type → Select T4 GPU and click Save.
  3. Copy and paste each cell below into separate Colab cells in sequential order.
  4. Run all cells (or press Shift + Enter for each cell).

CELL 3 High-Resolution Patch Extraction Dataset (Prevents Colab RAM Crash)
Extracts dynamic 128×128 crops from full 2K images at runtime to maximize texture diversity while keeping RAM usage well under Colab's 12 GB limit.
# Cell 3: Memory-Efficient Large-Scale Patch Dataset
import os
import glob
import random
import torch
from torch.utils.data import Dataset, DataLoader
from PIL import Image
import torchvision.transforms.functional as TF

class LargeScalePatchDataset(Dataset):
    def __init__(self, img_dir, patch_size=128, patches_per_img=2):
        self.img_paths = sorted(glob.glob(os.path.join(img_dir, "*.png")) + glob.glob(os.path.join(img_dir, "*.jpg")))
        self.patch_size = patch_size
        self.patches_per_img = patches_per_img

    def __len__(self):
        return len(self.img_paths) * self.patches_per_img

    def __getitem__(self, idx):
        img_idx = idx // self.patches_per_img
        path = self.img_paths[img_idx]
        
        try:
            img = Image.open(path).convert('L') # Luminance (Y) channel
        except Exception:
            fallback_idx = random.randint(0, len(self.img_paths) - 1)
            img = Image.open(self.img_paths[fallback_idx]).convert('L')
            
        w, h = img.size
        if w < self.patch_size or h < self.patch_size:
            img = img.resize((max(w, self.patch_size), max(h, self.patch_size)), Image.BICUBIC)
            w, h = img.size

        # Random Crop to multiple of 8
        top = random.randint(0, h - self.patch_size)
        left = random.randint(0, w - self.patch_size)
        patch = TF.crop(img, top, left, self.patch_size, self.patch_size)
        
        # Data Augmentation: Horizontal and Vertical Flips
        if random.random() > 0.5:
            patch = TF.hflip(patch)
        if random.random() > 0.5:
            patch = TF.vflip(patch)

        return TF.to_tensor(patch) # [1, 128, 128] in [0, 1]

train_dataset = LargeScalePatchDataset("data/train_hr", patch_size=128, patches_per_img=2)
train_loader = DataLoader(train_dataset, batch_size=16, shuffle=True, num_workers=4, pin_memory=True)
print(f"Dataset Pipeline ready: {len(train_dataset):,} patches generated per epoch.")
CELL 4 Train Frequency GNN (25 Epochs)
Trains the Frequency GNN on 72-node spatial-frequency graphs with 10-channel features using Frequency-Weighted L1 Loss.
# Cell 4: Train Stage 1 Frequency GNN (25 Epochs)
import torch
import torch.nn as nn
from src.models import FrequencyGNN, FrequencyWeightedL1Loss
from src.dataset import compute_block_dct_and_features
import os

device = torch.device('cuda' if torch.cuda.is_available() else 'cpu')
os.makedirs("checkpoints", exist_ok=True)

# 1. Initialize Frequency GNN Model
gnn_model = FrequencyGNN(in_channels=10, hidden_channels=128, num_layers=4).to(device)
optimizer_gnn = torch.optim.Adam(gnn_model.parameters(), lr=1e-3, weight_decay=1e-5)
scheduler_gnn = torch.optim.lr_scheduler.CosineAnnealingLR(optimizer_gnn, T_max=25, eta_min=1e-5)
criterion_gnn = FrequencyWeightedL1Loss(cutoff=15).to(device)

print("=" * 60)
print("STAGE 1: Training Frequency GNN for 25 Epochs on DF2K...")
print(f"Device: {device} | Total Parameters: {sum(p.numel() for p in gnn_model.parameters() if p.requires_grad):,}")
print("=" * 60)

for epoch in range(1, 26):
    gnn_model.train()
    running_loss = 0.0
    steps = 0
    
    for batch_imgs in train_loader:
        batch_imgs = batch_imgs.to(device) # [B, 1, 128, 128]
        
        # Compute vectorized multi-block DCT & Graph features
        x_nodes, edge_index, edge_attr, target_ac = compute_block_dct_and_features(batch_imgs, cutoff=15)
        x_nodes, edge_index = x_nodes.to(device), edge_index.to(device)
        target_ac = target_ac.to(device)
        
        optimizer_gnn.zero_grad()
        pred_ac = gnn_model(x_nodes, edge_index)
        loss = criterion_gnn(pred_ac, target_ac)
        
        loss.backward()
        torch.nn.utils.clip_grad_norm_(gnn_model.parameters(), 1.0)
        optimizer_gnn.step()
        
        running_loss += loss.item()
        steps += 1
        if steps >= 300: # 300 batches per epoch keeps training brisk on large datasets
            break
            
    scheduler_gnn.step()
    avg_loss = running_loss / max(steps, 1)
    print(f"GNN Epoch [{epoch:02d}/25] | Frequency L1 Loss: {avg_loss:.5f} | LR: {scheduler_gnn.get_last_lr()[0]:.6f}")

# Save GNN Checkpoint
torch.save(gnn_model.state_dict(), "checkpoints/model.pth")
print("✅ Stage 1 GNN weights saved to: checkpoints/model.pth")
CELL 5 Train Spatial Deblocking CNN (60 Epochs)
Trains the Spatial Refiner CNN for 60 epochs to eliminate 8×8 JPEG block seams and restore spatial continuous edges.
# Cell 5: Train Stage 2 Spatial Refiner CNN (60 Epochs)
import torch
import torch.nn as nn
from src.models import SpatialRefinerCNN
from src.dataset import simulate_jpeg_truncation

cnn_model = SpatialRefinerCNN(in_channels=1, hidden_channels=64, num_residual_blocks=6).to(device)
optimizer_cnn = torch.optim.Adam(cnn_model.parameters(), lr=5e-4, weight_decay=1e-5)
scheduler_cnn = torch.optim.lr_scheduler.CosineAnnealingLR(optimizer_cnn, T_max=60, eta_min=1e-5)
criterion_cnn = nn.L1Loss().to(device)

print("=" * 60)
print("STAGE 2: Training Spatial Refiner CNN for 60 Epochs on DF2K...")
print(f"Device: {device} | Total Parameters: {sum(p.numel() for p in cnn_model.parameters() if p.requires_grad):,}")
print("=" * 60)

for epoch in range(1, 61):
    cnn_model.train()
    running_loss = 0.0
    steps = 0
    
    for batch_imgs in train_loader:
        target_clean = batch_imgs.to(device) # [B, 1, 128, 128]
        
        # Simulate baseline JPEG truncated input (Cutoff = 15)
        with torch.no_grad():
            degraded_input = simulate_jpeg_truncation(target_clean, cutoff=15)
            
        optimizer_cnn.zero_grad()
        restored = cnn_model(degraded_input)
        loss = criterion_cnn(restored, target_clean)
        
        loss.backward()
        torch.nn.utils.clip_grad_norm_(cnn_model.parameters(), 1.0)
        optimizer_cnn.step()
        
        running_loss += loss.item()
        steps += 1
        if steps >= 300:
            break
            
    scheduler_cnn.step()
    avg_loss = running_loss / max(steps, 1)
    if epoch % 5 == 0 or epoch == 1:
        print(f"CNN Epoch [{epoch:02d}/60] | Spatial L1 Loss: {avg_loss:.5f} | LR: {scheduler_cnn.get_last_lr()[0]:.6f}")

# Save CNN Checkpoint
torch.save(cnn_model.state_dict(), "checkpoints/spatial_refiner.pth")
print("✅ Stage 2 Spatial CNN weights saved to: checkpoints/spatial_refiner.pth")
